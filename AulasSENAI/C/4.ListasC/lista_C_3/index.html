2. Escreva um programa que escreva na tela, de 1 até 100, de 1 em 1, 3 vezes. A
primeira vez deve usar a estrutura de repetição for, a segunda while, e a terceira do
while.

#include 
#include

    int i;

    printf(" 1. Usando a estrutura FOR \n");
    for (i = 1; i <= 100; i++) {
        printf("%d ", i);
    }
    printf("\n\n");

  
    printf(" 2. Usando a estrutura WHILE \n");
    i = 1; 
    while (i <= 100) {
        printf("%d ", i);
        i++; 
    }
    printf("\n\n");
    printf(" 3. Usando a estrutura DO-WHILE \n");
    i = 1; 
    do {
        printf("%d ", i);
        i++;
    } while (i <= 100);
    printf("\n");
    return 0;
}

        long long t = b;
        b = a % b;
        a = t;
