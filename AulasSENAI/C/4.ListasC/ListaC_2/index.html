   EXERCÍCIO 02: Raiz quadrada de positivo
   ========================================================================== */
#include 
#include 

float raiz_quadrada_ex2(float n) {
    float x = n, y = 1.0, e = 0.00001;
    while (x - y > e) {
        x = (x + y) / 2.0;
        y = n / x;
    }
    return x;
}

int main() {
    float num;
    printf("Digite um numero: ");
    scanf("%f", &num);

    if (num >= 0) {
        printf("A raiz quadrada de %.2f e: %.2f\n", num, raiz_quadrada_ex2(num));
    } else {
        printf("Numero invalido.\n");
    }
    return 0;
}

        printf("Numero invalido\n");

        float x = (float)num;
        float y = (x - 1.0f) / (x + 1.0f);

/* ==========================================================================
   EXERCÍCIO 25: Equação do 2º grau
   ========================================================================== */
#include 
#include 

float raiz_quadrada_ex25(float n) {
    if (n <= 0) return 0;
    float x = n, y = 1.0, e = 0.00001;
    while (x - y > e) { x = (x + y) / 2.0; y = n / x; }
    return x;
}

int main() {
    float a, b, c, delta, x1, x2;
    printf("Digite os coeficientes a, b e c: ");
    scanf("%f %f %f", &a, &b, &c);

    if (a == 0) {
        printf("Nao e equacao de segundo grau\n");
    } else {
        delta = (b * b) - (4 * a * c);
        if (delta < 0) {
            printf("Nao existe raiz\n");
        } else if (delta == 0) {
            x1 = -b / (2 * a);
            printf("Raiz unica: %.2f\n", x1);
        } else {
            x1 = (-b + raiz_quadrada_ex25(delta)) / (2 * a);
            x2 = (-b - raiz_quadrada_ex25(delta)) / (2 * a);
            printf("Raizes: x1 = %.2f e x2 = %.2f\n", x1, x2);
        }
    }
    return 0;
}

/* ==========================================================================
   EXERCÍCIO 28: Cálculo de diferentes médias
   ========================================================================== */
#include 
#include 

float raiz_cubica_ex28(float n) {
    float x = n;
    for (int i = 0; i < 100; i++) x = (2.0f * x + n / (x * x)) / 3.0f;
    return x;
}

int main() {
    int x, y, z, op;
    printf("Digite 3 inteiros positivos: ");
    scanf("%d %d %d", &x, &y, &z);

    printf("1 - Geometrica\n2 - Ponderada\n3 - Harmonica\n4 - Aritmetica\nOpcao: ");
    scanf("%d", &op);

    switch (op) {
        case 1: printf("Geometrica: %.2f\n", raiz_cubica_ex28(x * y * z)); break;
        case 2: printf("Ponderada: %.2f\n", (x + 2.0*y + 3.0*z) / 6.0); break;
        case 3: printf("Harmonica: %.2f\n", 1.0 / (1.0/x + 1.0/y + 1.0/z)); break;
        case 4: printf("Aritmetica: %.2f\n", (x + y + z) / 3.0); break;
        default: printf("Opcao invalida\n");
    }
    return 0;
}

/* ==========================================================================
   EXERCÍCIO 37: Cálculo de tarifa de estacionamento
   ========================================================================== */
#include 
#include 

int main() {
    int h_ent, m_ent, h_sai, m_sai;
    int min_ent, min_sai, duracao_min, horas;
    float total = 0.0;

    printf("Digite a hora e minuto de chegada (ex: 12 50): ");
    scanf("%d %d", &h_ent, &m_ent);
    printf("Digite a hora e minuto de partida (ex: 14 10): ");
    scanf("%d %d", &h_sai, &m_sai);

    min_ent = h_ent * 60 + m_ent;
    min_sai = h_sai * 60 + m_sai;

    if (min_sai < min_ent) min_sai += 24 * 60; // Virada de dia

    duracao_min = min_sai - min_ent;
    horas = (duracao_min + 59) / 60; // Arredonda qualquer fração para cima

    if (horas <= 2) total = horas * 1.00;
    else if (horas <= 4) total = 2 * 1.00 + (horas - 2) * 1.40;
    else total = 2 * 1.00 + 2 * 1.40 + (horas - 4) * 2.00;

    printf("Horas cobradas: %dh | Total a pagar: R$ %.2f\n", horas, total);
    return 0;
}

    }

}
