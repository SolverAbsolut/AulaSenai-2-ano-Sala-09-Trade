                /* EXERCÍCIO 49: Horário de término de um experimento biológico */
                int ex49() {
                int h_inicio, m_inicio, s_inicio, duracao;
                int total_inicio, total_fim;
                int h_fim, m_fim, s_fim;

                printf("Digite o horario de inicio (horas minutos segundos): ");
                scanf("%d %d %d", &h_inicio, &m_inicio, &s_inicio);
                printf("Digite a duracao da experiencia em segundos: ");
                scanf("%d", &duracao);

                total_inicio = (h_inicio * 3600) + (m_inicio * 60) + s_inicio;
                total_fim = total_inicio + duracao;

                h_fim = (total_fim / 3600) % 24;
                m_fim = (total_fim % 3600) / 60;
                s_fim = total_fim % 60;

                printf("Horario de termino: %02dh:%02dm:%02ds\n", h_fim, m_fim, s_fim);
                return 0;
                }
